📝 Easy Integration

Add this to your website:

<!-- Include CSS -->
<link rel="stylesheet" href="https://your-domain.com/static/css/voice-widget.css">

<!-- Include JS -->
<script src="https://your-domain.com/static/js/voice-widget.js"></script>

<!-- Initialize Text Chat Widget -->
<script>
  WeeVoiceWidget.init({
    mode: 'text', // Set to 'text' for chat mode
    apiUrl: 'https://your-domain.com',
    agentId: YOUR_AGENT_ID,
    agentName: 'Your Assistant',
    greeting: 'Hello! How can I help you today?',
    color: '#4F46E5',
    position: 'bottom-right',
    language: 'en'
  });
</script>
